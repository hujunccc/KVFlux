cmake_minimum_required(VERSION 3.20)

project(KVFlux VERSION 1.10.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(KVFLUX_BUILD_TESTS "Build KVFlux tests" OFF)
option(KVFLUX_BUILD_BENCHMARKS "Build KVFlux benchmarks" OFF)
option(KVFLUX_BUILD_EXAMPLES "Build KVFlux examples" ON)
option(KVFLUX_ENABLE_CUDA "Build the real CUDA GPU memory pool" OFF)

add_library(kvflux src/block_manager.cpp src/physical_block_pool.cpp
                   src/v2/block_table.cpp src/v2/sequence_state.cpp src/v2/slot_mapping.cpp
                   src/v2/paged_kv_layout.cpp src/v2/paged_kv_read_table.cpp
                   src/v2/reference_attention.cpp
                   src/kv_layout.cpp src/metrics.cpp)
add_library(KVFlux::kvflux ALIAS kvflux)
target_include_directories(kvflux PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(kvflux PUBLIC cxx_std_17)
if(KVFLUX_ENABLE_CUDA)
  enable_language(CUDA)
  set(CMAKE_CUDA_STANDARD 17)
  set(CMAKE_CUDA_STANDARD_REQUIRED ON)
  find_package(CUDAToolkit REQUIRED)
  target_sources(kvflux PRIVATE src/gpu_memory_pool.cpp src/pinned_memory.cpp
                               src/cuda_stream.cpp src/async_transfer.cpp src/tiered_block_manager.cpp
                               src/v2/paged_kv_storage.cpp src/v2/paged_kv_write.cu
                               src/v2/paged_kv_read.cu src/v2/paged_attention.cu)
  target_link_libraries(kvflux PUBLIC CUDA::cudart)
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(kvflux PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic>)
endif()

if(KVFLUX_BUILD_EXAMPLES)
  add_executable(kvflux_demo examples/basic.cpp)
  target_link_libraries(kvflux_demo PRIVATE KVFlux::kvflux)
  add_executable(kvflux_capacity examples/capacity.cpp)
  target_link_libraries(kvflux_capacity PRIVATE KVFlux::kvflux)
  add_executable(kvflux_physical_pool_demo examples/physical_block_pool.cpp)
  target_link_libraries(kvflux_physical_pool_demo PRIVATE KVFlux::kvflux)
  add_executable(kvflux_block_table_demo examples/block_table.cpp)
  target_link_libraries(kvflux_block_table_demo PRIVATE KVFlux::kvflux)
  add_executable(kvflux_sequence_state_demo examples/sequence_state.cpp)
  target_link_libraries(kvflux_sequence_state_demo PRIVATE KVFlux::kvflux)
  add_executable(kvflux_dynamic_growth_demo examples/dynamic_sequence_growth.cpp)
  target_link_libraries(kvflux_dynamic_growth_demo PRIVATE KVFlux::kvflux)
  add_executable(kvflux_slot_mapping_demo examples/slot_mapping.cpp)
  target_link_libraries(kvflux_slot_mapping_demo PRIVATE KVFlux::kvflux)
  if(KVFLUX_ENABLE_CUDA)
    add_executable(kvflux_paged_kv_storage_demo examples/paged_kv_storage.cpp)
    target_link_libraries(kvflux_paged_kv_storage_demo PRIVATE KVFlux::kvflux)
    add_executable(kvflux_paged_kv_write_demo examples/paged_kv_write.cpp)
    target_link_libraries(kvflux_paged_kv_write_demo PRIVATE KVFlux::kvflux)
    add_executable(kvflux_paged_kv_read_demo examples/paged_kv_read.cpp)
    target_link_libraries(kvflux_paged_kv_read_demo PRIVATE KVFlux::kvflux)
    add_executable(kvflux_gpu_demo examples/gpu_pool.cpp)
    target_link_libraries(kvflux_gpu_demo PRIVATE KVFlux::kvflux)
    add_executable(kvflux_async_demo examples/async_transfer.cpp)
    target_link_libraries(kvflux_async_demo PRIVATE KVFlux::kvflux)
    add_executable(kvflux_tiered_demo examples/tiered_cache.cpp)
    target_link_libraries(kvflux_tiered_demo PRIVATE KVFlux::kvflux)
  endif()
endif()

if(KVFLUX_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()

if(KVFLUX_BUILD_BENCHMARKS)
  add_subdirectory(benchmark)
endif()
